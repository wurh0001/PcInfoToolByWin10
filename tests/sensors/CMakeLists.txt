cmake_minimum_required(VERSION 3.20)
project(PcInfoAmdTemperatureProbe LANGUAGES CXX)
add_executable(PcInfoTool-DiskCapacityTest physical_disk_capacity_test.cpp ../../src/core/collectors/PhysicalDiskSampler.cpp)
target_include_directories(PcInfoTool-DiskCapacityTest PRIVATE ../../src)
target_compile_features(PcInfoTool-DiskCapacityTest PRIVATE cxx_std_20)
target_compile_options(PcInfoTool-DiskCapacityTest PRIVATE /utf-8)
target_compile_definitions(PcInfoTool-DiskCapacityTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(PcInfoTool-DiskCapacityTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../build")
